# 知识库的 Python 依赖。
#
# 装法一（推荐，脚本会自动做这套）：
#     scripts\install-env.cmd
#
# 装法二手动：
#     py -3 -m venv .venv
#     .venv\Scripts\python.exe -m pip install -r requirements.txt ^
#         -i https://mirrors.aliyun.com/pypi/simple/
#
# ⚠ 只列**直接依赖**，不锁传递依赖的版本：换成别人的 Python 版本或别的
#   操作系统时，pip 才能挑到能装的二进制 wheel。要完全复现本机环境，
#   用 `pip freeze > requirements.lock.txt` 另存一份。
#
# ⚠ 国内必须换源：实测清华源（TUNA）上找不到 mcp 包，阿里源可以。
#   嵌入模型走 hf-mirror.com（convert.py 里已设 HF_ENDPOINT）。

pymupdf>=1.24

# 向量计算（searcher.py / converter.py 直接 import numpy）

mcp>=1.2

zstandard>=0.22

# 管理面板显示头像（tools/gui.py 的 _load_avatar）。
